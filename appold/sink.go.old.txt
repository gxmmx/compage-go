package app

import (
	"context"
	"sync"
)

// -----------------------------------------------------------------------------
// Interfaces
// -----------------------------------------------------------------------------

type UnitSink interface {
	get() (value any, isset bool)
	set(value any)
}

// -----------------------------------------------------------------------------
// Concrete types
// -----------------------------------------------------------------------------

type Sink struct {
	mu    sync.RWMutex
	data  any
	isSet bool
}

// -----------------------------------------------------------------------------
// Constructors
// -----------------------------------------------------------------------------

func NewSink() *Sink {
	return &Sink{
		mu:    sync.RWMutex{},
		data:  nil,
		isSet: false,
	}
}

func WaitForSink[T any](ctx context.Context, s UnitSink) (value T, isset bool) {
	v, isset := s.get()
	if !isset {
		return *new(T), isset
	}
	return v.(T), isset
}

// -----------------------------------------------------------------------------
// Methods
// -----------------------------------------------------------------------------

// Set stores a typed value safely
func (s *Sink) set(v any) {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.data = v
	s.isSet = true
}

// Get retrieves the typed value safely
func (s *Sink) get() (value any, isset bool) {
	s.mu.RLock()
	defer s.mu.RUnlock()
	return s.data, s.isSet
}
