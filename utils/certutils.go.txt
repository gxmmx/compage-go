// -----------------------------------------------------------------------------
// Byte parsing functions
// -----------------------------------------------------------------------------

func CertificatesToByteSlice(crts []*x509.Certificate) [][]byte {
	var certBytes [][]byte
	for _, cert := range crts {
		certBytes = append(certBytes, cert.Raw)
	}
	return certBytes
}

func ParseKeyFromString(content string) (crypto.PrivateKey, error) {
	blocks, err := extractPemBlocks(content)
	if err != nil {
		return nil, err
	}
	return parseKeyBlock(blocks[0])
}

	}

// String Bundle Parsing functions
// -----------------------------------------------------------------------------

// Parse certificates and keys from a string
func ParseCrtBundleFromString(content string) ([]*x509.Certificate, []crypto.PrivateKey, error) {
	blocks, err := extractPemBlocks(content)
	if err != nil {
		return nil, nil, err
	}

	var certs []*x509.Certificate
	var keys []crypto.PrivateKey

	for _, block := range blocks {
		switch block.Type {
		case "CERTIFICATE":
			var cert *x509.Certificate
			var err error
			cert, err = x509.ParseCertificate(block.Bytes)
			if err != nil {
				return nil, nil, fmt.Errorf("failed to parse certificate: %w", err)
			}
			certs = append(certs, cert)
		case "PRIVATE KEY", "RSA PRIVATE KEY":
			var key crypto.PrivateKey
			var err error
			// PKCS8
			key, err = x509.ParsePKCS8PrivateKey(block.Bytes)
			if err != nil {
				//PKCS1
				key, err = x509.ParsePKCS1PrivateKey(block.Bytes)
				if err != nil {
					// EC
					key, err = x509.ParseECPrivateKey(block.Bytes)
					if err != nil {
						return nil, nil, errors.New("invalid private key format")
					}
				}
			}
			// ED25519
			if edKey, ok := key.(ed25519.PrivateKey); ok {
				key = edKey
			}
			keys = append(keys, key)
		default:
			return nil, nil, fmt.Errorf("invalid PEM block type: %s", block.Type)
		}
	}

	return certs, keys, nil
}

func ParseCrtBundleFromFile(path string) ([]*x509.Certificate, []crypto.PrivateKey, error) {
	data, err := FetchStringFromFile(path)
	if err != nil {
		return nil, nil, err
	}
	return ParseCrtBundleFromString(data)
}

// Validates a certificate and private key pair.
func ValidateKeyPair(cert *x509.Certificate, key crypto.PrivateKey) error {
	switch key.(type) {
	case *rsa.PrivateKey:
		certPubKey, ok := cert.PublicKey.(*rsa.PublicKey)
		if !ok {
			return errors.New("certificate public key is not a valid")
		}
		if certPubKey.N.Cmp(key.(*rsa.PrivateKey).PublicKey.N) != 0 || certPubKey.E != key.(*rsa.PrivateKey).PublicKey.E {
			return errors.New("private key does not match certificate")
		}
		return nil
	case ed25519.PrivateKey:
		certPubKey, ok := cert.PublicKey.(ed25519.PublicKey)
		if !ok {
			return errors.New("certificate public key is not a valid")
		}
		if certPubKey.Equal(key.(ed25519.PrivateKey).Public().(ed25519.PublicKey)) {
			return errors.New("private key does not match certificate")
		}
		return nil
	case *ecdsa.PrivateKey:
		certPubKey, ok := cert.PublicKey.(*ecdsa.PublicKey)
		if !ok {
			return errors.New("certificate public key is not a valid")
		}
		if certPubKey.X.Cmp(key.(*ecdsa.PrivateKey).PublicKey.X) != 0 || certPubKey.Y.Cmp(key.(*ecdsa.PrivateKey).PublicKey.Y) != 0 {
			return errors.New("private key does not match certificate")
		}
		return nil
	default:
		return errors.New("invalid private key type")
	}
}

// Validates a certificate expiry.
func ValidateCrtExpiry(crt *x509.Certificate) error {
	now := time.Now()
	if now.Before(crt.NotBefore) {
		return errors.New("certificate is not yet valid")
	}
	if now.After(crt.NotAfter) {
		return errors.New("certificate has expired")
	}
	return nil
}

// Validates a certificate chain.
func ValidateCrtChain(cert *x509.Certificate, chain []*x509.Certificate, rootCA *x509.Certificate) error {
	// Create one chain from certificate, chain and root CA
	var fullChain []*x509.Certificate
	if cert != nil {
		fullChain = append(fullChain, cert)
	}
	if len(chain) != 0 {
		fullChain = append(fullChain, chain...)
	}
	if rootCA != nil {
		fullChain = append(fullChain, rootCA)
	}

	if len(fullChain) < 2 {
		return nil
	}
	// Validate chain
	for i := 0; i < len(fullChain)-1; i++ {
		issuer := fullChain[i+1]
		cert := fullChain[i]
		// Check if duplicate certificates exist in chain
		if bytes.Equal([]byte(cert.Subject.CommonName), []byte(issuer.Subject.CommonName)) {
			return fmt.Errorf("duplicate certificate at index %d", i)
		}
		// Check if cert was issued by issuer
		if err := cert.CheckSignatureFrom(issuer); err != nil {
			if _, ok := err.(x509.ConstraintViolationError); ok {
				return fmt.Errorf("certificate at index %d is not issued by certificate at index %d due to constraint violation: %v", i, i+1, err)
			}
			return fmt.Errorf("certificate at index %d is not issued by certificate at index %d: %v", i, i+1, err)
		}
	}

	return nil
}

// -----------------------------------------------------------------------------
// Internal extract functions
// -----------------------------------------------------------------------------

// Extracts pem blocks from a string
func extractPemBlocks(content string) ([]*pem.Block, error) {
	// Check empty content
	if len(content) == 0 {
		return nil, fmt.Errorf("no pem blocks found")
	}
	// Check for base64 encoded content
	var rawdata []byte
	if !strings.Contains(content, "-----BEGIN") {
		decoded, err := base64.StdEncoding.DecodeString(content)
		if err != nil {
			return nil, fmt.Errorf("invalid pem content")
		}
		rawdata = decoded
	} else {
		rawdata = []byte(content)
	}
	data := bytes.ReplaceAll(rawdata, []byte("\\n"), []byte("\n"))

	// Decode pem blocks
	var blocks []*pem.Block
	for len(data) > 0 {
		var block *pem.Block
		block, data = pem.Decode(data)
		if block == nil {
			break
		}
		blocks = append(blocks, block)
	}

	if len(bytes.TrimSpace(data)) > 0 {
		return nil, fmt.Errorf("invalid pem content")
	}

	// if len(blocks) == 0 {
	// 	return nil, fmt.Errorf("no pem blocks found")
	// }
	return blocks, nil
}

// -----------------------------------------------------------------------------
// Internal parse functions
// -----------------------------------------------------------------------------

// Parses a certificate from a pem block
func parseCrtBlock(block *pem.Block) (*x509.Certificate, error) {
	var cert *x509.Certificate
	var err error
	switch block.Type {
	case "CERTIFICATE":
		cert, err = x509.ParseCertificate(block.Bytes)
		if err != nil {
			return nil, fmt.Errorf("failed to parse certificate: %w", err)
		}
	default:
		return nil, fmt.Errorf("invalid pem block type: %s", block.Type)
	}
	return cert, nil
}

// Parses a private key from a pem block
func parseKeyBlock(block *pem.Block) (crypto.PrivateKey, error) {
	var key crypto.PrivateKey
	var err error
	switch block.Type {
	case "PRIVATE KEY", "RSA PRIVATE KEY":
		// PKCS8
		key, err = x509.ParsePKCS8PrivateKey(block.Bytes)
		if err == nil {
			// ED25519
			if edKey, ok := key.(ed25519.PrivateKey); ok {
				return edKey, nil
			}
			return key, nil
		}
		// PKCS1
		if key, err = x509.ParsePKCS1PrivateKey(block.Bytes); err == nil {
			return key, nil
		}
		// // EC
		// if key, err = x509.ParseECPrivateKey(block.Bytes); err == nil {
		// 	return key, nil
		// }
		return nil, errors.New("invalid private key format")
	default:
		return nil, fmt.Errorf("invalid pem block type: %s", block.Type)
	}
}

// Parses a certificate request from a pem block
func parseCsrBlock(block *pem.Block) (*x509.CertificateRequest, error) {
	var csr *x509.CertificateRequest
	var err error
	switch block.Type {
	case "CERTIFICATE REQUEST":
		csr, err = x509.ParseCertificateRequest(block.Bytes)
		if err != nil {
			return nil, fmt.Errorf("failed to parse certificate request: %w", err)
		}
	default:
		return nil, fmt.Errorf("invalid pem block type: %s", block.Type)
	}
	return csr, nil
}
