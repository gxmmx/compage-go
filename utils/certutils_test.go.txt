// Helper function to create a rsa test stack with a self-signed certificate
func createTestStack() (*x509.Certificate, *rsa.PrivateKey, *x509.CertificateRequest, error) {
	privKey, err := GenerateRsaPrivateKey(2048)
	if err != nil {
		return nil, nil, nil, fmt.Errorf("failed to generate private key: %w", err)
	}
	subject := GenerateCertificateSubject(
		"some.domain.com",
		"TestOrg",
		"IT",
		"US",
		"California",
		"San Francisco",
	)
	csr, err := GenerateCsr(subject, privKey, nil, nil, nil, nil)
	if err != nil {
		return nil, nil, nil, fmt.Errorf("failed to generate CSR: %w", err)
	}
	cert, err := GenerateSelfsignedCertificate(csr, privKey, 1)
	if err != nil {
		return nil, nil, nil, fmt.Errorf("failed to generate self-signed certificate: %w", err)
	}
	return cert, privKey, csr, nil
}

func TestKey(t *testing.T) {
	t.Run("empty and invalid", func(t *testing.T) {
		pemStr := EncodeKeyToPem(nil, false, false)
		if pemStr != "" {
			t.Errorf("expected empty string, got: %q", pemStr)
		}
		unsupportedKey, err := ecdsa.GenerateKey(elliptic.P256(), rand.Reader)
		if err != nil {
			t.Fatalf("unexpected error: %v", err)
		}
		pemStr = EncodeKeyToPem(unsupportedKey, false, false)
		if pemStr != "" {
			t.Errorf("expected empty string, got: %q", pemStr)
		}
		_, err = ParseKeyFromString("")
		if err == nil {
			t.Errorf("expected error, got nil")
		}
		tmpDir := t.TempDir()
		keyFile, err := os.CreateTemp(tmpDir, "empty-*.key")
		if err != nil {
			t.Fatalf("unexpected error: %v", err)
		}
		defer os.Remove(keyFile.Name())
		keyFile.WriteString("")
		keyFile.Close()
		_, err = ParseKeyFromFile(keyFile.Name())
		if err == nil {
			t.Errorf("expected error, got nil")
		}
		// Non-existing file
		_, err = ParseKeyFromFile(keyFile.Name() + ".nonexisting")
		if err == nil {
			t.Errorf("expected error, got nil")
		}
		// Invalid strings
		_, err = ParseKeyFromString("-----BEGIN PRIVATE KEY-----\ninvalidpemblock\n-----END PRIVATE KEY-----\n")
		if err == nil {
			t.Errorf("expected error, got nil")
		}
		_, err = ParseKeyFromString("notevenapemblock$")
		if err == nil {
			t.Errorf("expected error, got nil")
		}
		_, err = ParseKeyFromString("-----BEGIN UNKNOWN-----\nfoo\n-----END UNKNOWN-----")
		if err == nil {
			t.Errorf("expected error, got nil")
		}
	})
	t.Run("rsa", func(t *testing.T) {
		// Generate a new RSA private key
		key, err := GenerateRsaPrivateKey(2048)
		if err != nil {
			t.Fatalf("unexpected error: %v", err)
		}

		t.Run("multiline", func(t *testing.T) {
			// Encode to multiline PEM format
			keyStr := EncodeKeyToPem(key, false, false)
			if !strings.Contains(keyStr, "-----BEGIN RSA PRIVATE KEY-----") {
				t.Errorf("expected PEM header, got: %q", keyStr)
			}
			if strings.Contains(keyStr, "\\n") {
				t.Errorf("expected multiline output, got: %q", keyStr)
			}
			if _, err := ParseKeyFromString(keyStr); err != nil {
				t.Fatalf("failed to parse PEM string: %v", err)
			}
			// Save to a temporary file
			tmpDir := t.TempDir()
			keyFile, err := os.CreateTemp(tmpDir, "rsa-*.key")
			if err != nil {
				t.Fatalf("unexpected error: %v", err)
			}
			defer os.Remove(keyFile.Name())
			keyFile.WriteString(keyStr)
			keyFile.Close()
			// Read from the file
			keyFromFile, err := ParseKeyFromFile(keyFile.Name())
			if err != nil {
				t.Fatalf("unexpected error: %v", err)
			}
			if keyFromFile == nil {
				t.Errorf("expected non-nil key, got nil")
			}
			if !reflect.DeepEqual(keyFromFile, key) {
				t.Errorf("parsed key does not match the original key")
			}
		})
		t.Run("singleline", func(t *testing.T) {
			// Encode to single line PEM format
			pemStr := EncodeKeyToPem(key, true, false)
			if !strings.Contains(pemStr, "-----BEGIN RSA PRIVATE KEY-----") {
				t.Errorf("expected PEM header, got: %q", pemStr)
			}
			if !strings.Contains(pemStr, "\\n") {
				t.Errorf("expected single line output, got: %q", pemStr)
			}
			// Decode the PEM string
			_, err := ParseKeyFromString(pemStr)
			if err != nil {
				t.Fatalf("failed to parse PEM string: %v", err)
			}
		})
		t.Run("B64 multiline", func(t *testing.T) {
			// Encode to base64 single line PEM format
			pemStr := EncodeKeyToPem(key, false, true)
			decoded, err := base64.StdEncoding.DecodeString(pemStr)
			if err != nil {
				t.Fatalf("failed to decode base64: %v", err)
			}
			if !strings.Contains(string(decoded), "-----BEGIN RSA PRIVATE KEY-----") {
				t.Errorf("expected PEM header, got: %q", pemStr)
			}
			if strings.Contains(string(decoded), "\\n") {
				t.Errorf("expected multiline output, got: %q", string(decoded))
			}
			if _, err := ParseKeyFromString(pemStr); err != nil {
				t.Fatalf("failed to parse PEM string: %v", err)
			}
		})
		t.Run("B64 singleline", func(t *testing.T) {
			// Encode to base64 multiline PEM format
			pemStr := EncodeKeyToPem(key, true, true)
			decoded, err := base64.StdEncoding.DecodeString(pemStr)
			if err != nil {
				t.Fatalf("failed to decode base64: %v", err)
			}
			if !strings.Contains(string(decoded), "-----BEGIN RSA PRIVATE KEY-----") {
				t.Errorf("expected PEM header, got: %q", pemStr)
			}
			if !strings.Contains(string(decoded), "\\n") {
				t.Errorf("expected single line output, got: %q", string(decoded))
			}
			if _, err := ParseKeyFromString(pemStr); err != nil {
				t.Fatalf("failed to parse PEM string: %v", err)
			}
		})
	})
	t.Run("ed25519", func(t *testing.T) {
		// Generate a new RSA private key
		key, err := GenerateEd25519PrivateKey()
		if err != nil {
			t.Fatalf("unexpected error: %v", err)
		}
		t.Run("multiline", func(t *testing.T) {
			// Encode to multiline PEM format
			keyStr := EncodeKeyToPem(key, false, false)
			if !strings.Contains(keyStr, "-----BEGIN PRIVATE KEY-----") {
				t.Errorf("expected PEM header, got: %q", keyStr)
			}
			if strings.Contains(keyStr, "\\n") {
				t.Errorf("expected multiline output, got: %q", keyStr)
			}
			if _, err := ParseKeyFromString(keyStr); err != nil {
				t.Fatalf("failed to parse PEM string: %v", err)
			}
			// Save to a temporary file
			tmpDir := t.TempDir()
			keyFile, err := os.CreateTemp(tmpDir, "ed25519-*.key")
			if err != nil {
				t.Fatalf("unexpected error: %v", err)
			}
			defer os.Remove(keyFile.Name())
			keyFile.WriteString(keyStr)
			keyFile.Close()
			// Read from the file
			keyFromFile, err := ParseKeyFromFile(keyFile.Name())
			if err != nil {
				t.Fatalf("unexpected error: %v", err)
			}
			if keyFromFile == nil {
				t.Errorf("expected non-nil key, got nil")
			}
			if !reflect.DeepEqual(keyFromFile, key) {
				t.Errorf("parsed key does not match the original key")
			}
		})
	})
}
